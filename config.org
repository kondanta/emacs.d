    "Kill the `thing-at-point' for the specified kind of THING."
    (let ((bounds (bounds-of-thing-at-point thing)))
      (if bounds
	  (kill-region (car bounds) (cdr bounds))
	(error "No %s at point" thing))))

    "Kill the word at point."
